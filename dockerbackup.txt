version: '3.8'

services:
  postgres:
    image: garapadev/postgres-postgis-pgvector
    container_name: bookmytree_postgres
    environment:
      POSTGRES_DB: bookmytree_master_db
      POSTGRES_USER: postgres
      POSTGRES_PASSWORD: secretpassword
    ports:
      - "5432:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - bookmytree-net

  redis:
    image: redis:7.2-alpine
    container_name: bookmytree_redis
    ports:
      - "6379:6379"
    networks:
      - bookmytree-net

  zookeeper:
    image: confluentinc/cp-zookeeper:7.5.0
    container_name: bookmytree_zookeeper
    environment:
      ZOOKEEPER_CLIENT_PORT: 2181
      ZOOKEEPER_TICK_TIME: 2000
    networks:
      - bookmytree-net

  kafka:
    image: confluentinc/cp-kafka:7.5.0
    container_name: bookmytree_kafka
    depends_on:
      - zookeeper
    ports:
      - "9092:9092"
    environment:
      KAFKA_BROKER_ID: 1
      KAFKA_ZOOKEEPER_CONNECT: 'zookeeper:2181'
      KAFKA_ADVERTISED_LISTENERS: PLAINTEXT://localhost:9092,PLAINTEXT_HOST://localhost:29092
      KAFKA_LISTENER_SECURITY_PROTOCOL_MAP: PLAINTEXT:PLAINTEXT,PLAINTEXT_HOST:PLAINTEXT
      KAFKA_INTER_BROKER_LISTENER_NAME: PLAINTEXT
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
    networks:
      - bookmytree-net

volumes:
  postgres_data:
    driver: local

networks:
  bookmytree-net:
    driver: bridge
